/* उपलब्ध (शिल्लक) असलेल्या M.B. ची यादी तयार करणे */ function populateAvailableMBs() { var size = document.getElementById('newMBSize').value; var sel = document.getElementById('selectAvailableMB'); sel.innerHTML = ''; var allRecords = getData('mb_records'); // आधीच वाटप केलेल्या (दिलेल्या) सर्व M.B. क्रमांकांची यादी var alreadyGivenMBs = allRecords.filter(function(m) { return m.actionType === 'M.B. दिली'; }).map(function(m) { return m.mbNumber; }); // केवळ मिळालेल्या आणि अजून कोणालाही न दिलेल्या, संबंधित आकाराच्या वह्या फिल्टर करणे var availableList = allRecords.filter(function(m) { return m.actionType === 'M.B. मिळाली' && m.size === size && !alreadyGivenMBs.includes(m.mbNumber); }); if (availableList.length === 0) { sel.add(new Option('या आकारात कोणतीही M.B. उपलब्ध नाही', '')); } else { availableList.forEach(function(m) { sel.add(new Option(m.mbNumber, m.mbNumber)); }); } } /* M.B. नोंद सेव्ह करताना दुबार नोंद तपासणी (Validation) */ function saveOrUpdateMB() { var actionType = document.getElementById('mbActionType').value; var size = document.getElementById('newMBSize').value; var no = ''; if (actionType === 'M.B. दिली') { no = document.getElementById('selectAvailableMB').value; } else { no = document.getElementById('newMBNo').value.trim(); } var src = document.getElementById('mbReceivedFromSource').value; var date = document.getElementById('newMBDate').value; var idx = parseInt(document.getElementById('editMBIndex').value, 10); if (!no || !src) return alert('कृपया M.B. क्रमांक आणि संबंधित व्यक्ती/कार्यालय निवडा.'); var sub = (src === 'JE' || src === 'विभागीय कार्यालय') ? document.getElementById('receivedSubSelect').value : ''; var partyDetail = src === 'चार्ज नुसार' ? 'चार्ज नुसार' : (src + (sub ? ' (' + sub + ')' : '')); var displayParty = actionType + ' : ' + partyDetail; var list = getData('mb_records'); // ==================== दुबार नोंद तपासणी (Duplicate Check) ==================== if (idx === -1) { // नवीन नोंद करताना if (actionType === 'M.B. मिळाली') { // १. ही M.B. आधीच कार्यालयात नोंदवली आहे का? var alreadyReceived = list.find(function(m) { return m.mbNumber.toLowerCase() === no.toLowerCase() && m.actionType === 'M.B. मिळाली'; }); if (alreadyReceived) { alert('⚠️ चूक: M.B. क्रमांक "' + no + '" ची नोंद आधीच कार्यालयात झालेली आहे!'); return; } } else if (actionType === 'M.B. दिली') { // २. ही M.B. आधीच कोणाला दिली आहे का? var alreadyGiven = list.find(function(m) { return m.mbNumber.toLowerCase() === no.toLowerCase() && m.actionType === 'M.B. दिली'; }); if (alreadyGiven) { alert('⚠️ चूक: M.B. क्रमांक "' + no + '" ही आधीच संबंधितांना (' + (alreadyGiven.receivedDetail || alreadyGiven.receivedSource) + ') देण्यात आलेली आहे!'); return; } } } // माहिती सेव्ह किंवा अपडेट करणे if (idx === -1) { list.push({ actionType: actionType, mbNumber: no, size: size, receivedSource: src, receivedDetail: sub, receivedFrom: displayParty, date: date }); // M.B. JE ला दिल्यास Assign मध्ये आपोआप नोंद होणे if (actionType === 'M.B. दिली' && src === 'JE' && sub) { var assigns = getData('assign_records'); assigns.push({ mbNo: no, jeName: sub, assignDate: date }); setData('assign_records', assigns); } } else { list[idx].actionType = actionType; list[idx].mbNumber = no; list[idx].size = size; list[idx].receivedSource = src; list[idx].receivedDetail = sub; list[idx].receivedFrom = displayParty; list[idx].date = date; } setData('mb_records', list); cancelMBEdit(); refreshAll(); alert('नोंद यशस्वीपणे जतन झाली!'); }
